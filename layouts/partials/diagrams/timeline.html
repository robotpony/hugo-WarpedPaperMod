{{/* Draws a ```timeline block, from render-codeblock-timeline.html by way of partials/diagram.html, which adds the source toggle; it takes the hook's context. Dots on a track of decades. The syntax (one "YEAR Label" per line, ~YEAR faint) is shared with the cookbook; see hugo-warped's fences/timeline.md. hugo-warped's parse.html reads the lines (notes dropped, bad lines warned) and this draws them. Ported from the cookbook's lineage timeline (not-a-chef, Fugu's blueprint/lineage.html), with its own .tline-* classes so the two never collide. The track runs from the first year's decade to the last year rounded up to a decade, with a tick at each decade and its year centred just under it (every other decade on tracks over 80 years). Labels alternate above and below the track by entry order. A label below that would cross a decade's year goes above instead, and the next one goes below. On each side a label takes the nearest row that its estimated width clears (Lora at 14px against a 600px track, the narrowest above 640px; inline code counts wider), with a leader to its dot. Labels near the ends align to their dot's side. Years closer than $gap draw in a zoomed span, on a wider band, with a short tick for each year. Under 640px common/timeline.css turns it into a vertical list with each year inline. */}} {{- $p := partial "hugo-warped/parse.html" . -}} {{- partial "hugo-warped/warn.html" (dict "hook" . "parse" $p) -}} {{- $entries := slice -}} {{- range $i, $l := cond $p.fallback slice $p.lines -}} {{- $label := $.Page.RenderString (dict "display" "inline") $l.label -}} {{- $code := findRE `[^<]*` $label -}} {{- $codeChars := 0 -}} {{- range $code }}{{ $codeChars = add $codeChars (countrunes (plainify .)) }}{{ end -}} {{- $chars := countrunes (plainify $label) -}} {{- $px := add (mul (float (sub $chars $codeChars)) 7.0) (mul (float $codeChars) 7.2) (mul (len $code) 10) -}} {{- $entries = $entries | append (dict "year" (int $l.year) "faint" (eq $l.marker "~") "label" $label "w" (div $px 6.0) "key" (add (mul (int $l.year) 1000) $i) ) -}} {{- end -}} {{- with sort $entries "key" -}} {{- /* Entries go in year order whatever the line order; lines with the same year keep theirs. */ -}} {{- $first := (index . 0).year -}} {{- $last := (index . (sub (len .) 1)).year -}} {{- $start := mul (math.Floor (div (float $first) 10)) 10 | int -}} {{- $end := mul (math.Ceil (div (float $last) 10)) 10 | int -}} {{- if le $end $start }}{{ $end = add $start 10 }}{{ end -}} {{- $span := float (sub $end $start) -}} {{- /* Clusters (hugo-warped's fences/timeline.md): neighbouring years closer than $gap (21px against 600px, enough for their short ticks' years) are one cluster, drawn $gap apart. The rest of the track keeps one scale, $k percent a year, shrunk to make room; a shrink can bring more neighbours inside the gap, so it settles over a few passes. */ -}} {{- $gap := 3.5 -}} {{- $ys := slice -}} {{- range . }}{{ if not (in $ys .year) }}{{ $ys = $ys | append .year }}{{ end }}{{ end -}} {{- $k := div 100.0 $span -}} {{- $close := slice -}} {{- range seq 4 -}} {{- $close = slice -}} {{- $zoomed := 0.0 }}{{ $zoomedYears := 0.0 -}} {{- range $i, $y := first (sub (len $ys) 1) $ys -}} {{- $dy := float (sub (index $ys (add $i 1)) $y) -}} {{- $c := lt (mul $dy $k) $gap -}} {{- $close = $close | append $c -}} {{- if $c }}{{ $zoomed = add $zoomed $gap }}{{ $zoomedYears = add $zoomedYears $dy }}{{ end -}} {{- end -}} {{- $k = div (sub 100.0 $zoomed) (sub $span $zoomedYears) -}} {{- end -}} {{- /* Each distinct year's position, and the zoomed spans (first and last year of each run of close neighbours). */ -}} {{- $at := dict -}} {{- $pos := mul (float (sub (index $ys 0) $start)) $k -}} {{- $zooms := slice }}{{ $from := -1 -}} {{- range $i, $y := $ys -}} {{- if gt $i 0 -}} {{- $dy := float (sub $y (index $ys (sub $i 1))) -}} {{- $pos = add $pos (cond (index $close (sub $i 1)) $gap (mul $dy $k)) -}} {{- end -}} {{- $at = merge $at (dict (string $y) $pos) -}} {{- $in := and (lt $i (len $close)) (index $close $i) -}} {{- if and $in (lt $from 0) }}{{ $from = $y }}{{ end -}} {{- if and (not $in) (ge $from 0) -}} {{- $zooms = $zooms | append (dict "from" $from "to" $y) }}{{ $from = -1 -}} {{- end -}} {{- end -}} {{- /* Inside a zoomed span each year gets a short tick ('95) in place of the decade ticks it covers; a decade tick within $gap of one is dropped too. */ -}} {{- $years := slice -}} {{- range $zooms -}} {{- $z := . -}} {{- range $ys -}} {{- if and (ge . $z.from) (le . $z.to) -}} {{- $years = $years | append (dict "pos" (index $at (string .)) "label" (printf "'%02d" (mod . 100))) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Decade years, centred on their ticks: a dot whose leader would cross one keeps its label above. $decHalf is half a year's width (11px mono against 600px), plus a little. A decade sits between the positions of the years either side of it. */ -}} {{- $decHalf := 3.0 -}} {{- $decades := slice -}} {{- $every := cond (gt $span 80.0) 20 10 -}} {{- range seq $start 10 $end -}} {{- $d := . -}} {{- $lo := $start }}{{ $loAt := 0.0 }}{{ $hi := $end }}{{ $hiAt := 100.0 -}} {{- range $ys -}} {{- if le . $d }}{{ $lo = . }}{{ $loAt = index $at (string .) }}{{ end -}} {{- if and (gt . $d) (eq $hi $end) }}{{ $hi = . }}{{ $hiAt = index $at (string .) }}{{ end -}} {{- end -}} {{- if eq $lo $d }}{{ $hiAt = $loAt }}{{ $hi = add $lo 1 }}{{ end -}} {{- $p := add $loAt (div (mul (float (sub $d $lo)) (sub $hiAt $loAt)) (float (sub $hi $lo))) -}} {{- $covered := false -}} {{- range $years }}{{ if lt (math.Abs (sub .pos $p)) $gap }}{{ $covered = true }}{{ end }}{{ end -}} {{- range $zooms }}{{ if and (gt $d .from) (lt $d .to) }}{{ $covered = true }}{{ end }}{{ end -}} {{- if not $covered -}} {{- $named := eq (mod (sub $d $start) $every) 0 -}} {{- $decades = $decades | append (dict "year" $d "pos" $p "named" $named) -}} {{- end -}} {{- end -}} {{- $up := slice }}{{ $down := slice -}} {{- $side := "up" -}} {{- $marks := slice -}} {{- range . -}} {{- $pos := index $at (string .year) -}} {{- if eq $side "down" -}} {{- range where $decades "named" true -}} {{- if and (ge $pos (sub .pos $decHalf)) (le $pos (add .pos $decHalf)) }}{{ $side = "up" }}{{ end -}} {{- end -}} {{- end -}} {{- $left := sub $pos (div .w 2) -}} {{- $edge := "" -}} {{- if lt $left 0.0 -}} {{- $edge = "start" }}{{ $left = sub $pos 1 -}} {{- else if gt (add $pos (div .w 2)) 100.0 -}} {{- $edge = "end" }}{{ $left = sub (add $pos 1) .w -}} {{- end -}} {{- $right := add $left .w -}} {{- /* The nearest row on this side whose last label ends before this one starts, or a new row. */ -}} {{- $rows := cond (eq $side "up") $up $down -}} {{- $row := len $rows -}} {{- range $r, $edgeAt := $rows -}} {{- if and (eq $row (len $rows)) (gt $left (add $edgeAt 2)) }}{{ $row = $r }}{{ end -}} {{- end -}} {{- $next := slice -}} {{- range $r, $edgeAt := $rows }}{{ $next = $next | append (cond (eq $r $row) $right $edgeAt) }}{{ end -}} {{- if eq $row (len $rows) }}{{ $next = $next | append $right }}{{ end -}} {{- if eq $side "up" }}{{ $up = $next }}{{ else }}{{ $down = $next }}{{ end -}} {{- $marks = $marks | append (merge . (dict "pos" $pos "side" $side "row" $row "edge" $edge)) -}} {{- $side = cond (eq $side "up") "down" "up" -}} {{- end -}} {{- end -}}
