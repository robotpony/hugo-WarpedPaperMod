{{/* Draws a ```flow block, from render-codeblock-flow.html by way of partials/diagram.html, which adds the source toggle; it takes the hook's context. Steps in a row joined by arrows, each label boxed with its note under it. The syntax is shared with the cookbook; see docs/flow.md. Modelled on the cookbook's formula diagrams (Fugu's render-codeblock-formula.html, its → mode), without icons. Markdown | in git Hugo → public/ caption: How a post gets to the server One step per line, `Label | note`, the note optional. A leading → or -> on lines after the first is dropped, since the hook draws the arrows. An optional `caption:` line, last, goes under the row. Labels and notes are inline Markdown. A line with more than one |, an empty label, or a step after the caption warns and isn't drawn. The row reads as one image to screen readers ("Markdown in git, then Hugo, then …"). Under 640px common/flow.css stacks it as a column with arrows pointing down. */}} {{- $steps := slice -}} {{- $caption := "" -}} {{- range $i, $line := split (strings.TrimSpace .Inner) "\n" -}} {{- $line = strings.TrimSpace $line -}} {{- if not $line }}{{ continue }}{{ end -}} {{- if hasPrefix $line "caption:" -}} {{- $caption = strings.TrimSpace (strings.TrimPrefix "caption:" $line) -}} {{- continue -}} {{- end -}} {{- if $caption -}} {{- warnf "flow: skipped %q after the caption (the caption goes last) in %s" $line $.Position -}} {{- continue -}} {{- end -}} {{- with findRESubmatch `^(?:→|->)\s+(.*)$` $line 1 -}} {{- if not $steps }}{{ warnf "flow: the first step has an arrow in %s" $.Position }}{{ end -}} {{- $line = index (index . 0) 1 -}} {{- end -}} {{- $f := split $line "|" -}} {{- $label := strings.TrimSpace (index $f 0) -}} {{- if or (gt (len $f) 2) (not $label) -}} {{- warnf "flow: skipped %q (want \"Label | note\") in %s" $line $.Position -}} {{- continue -}} {{- end -}} {{- $note := "" -}} {{- if eq (len $f) 2 }}{{ $note = strings.TrimSpace (index $f 1) }}{{ end -}} {{- $steps = $steps | append (dict "label" $label "note" $note) -}} {{- end -}} {{- if gt (len $steps) 5 -}} {{- warnf "flow: %d steps in %s; more than five crowd the row" (len $steps) .Position -}} {{- end -}} {{- with $steps -}} {{- $aria := slice -}} {{- range . }}{{ $aria = $aria | append (strings.TrimSpace (printf "%s %s" (plainify ($.Page.RenderString (dict "display" "inline") .label)) (plainify ($.Page.RenderString (dict "display" "inline") .note)))) }}{{ end -}} {{- else -}} {{- warnf "flow: no steps in %s" .Position -}} {{- end -}}
