{{- /* Page title with an optional inline icon. Front matter `icon` is a bundle resource name or a site URL ("/images/x.svg"); `iconFor` places it before the first occurrence of that phrase, otherwise it goes before the whole title. Wrapped in a span so flex title rows keep the icon beside its text. */ -}} {{- $title := .Title | htmlEscape }} {{- $src := "" }} {{- with .Params.icon }} {{- if or (hasPrefix . "/") (strings.Contains . "://") }}{{ $src = relURL . }} {{- else }}{{ with $.Resources.Get . }}{{ $src = .RelPermalink }}{{ end }} {{- end }} {{- end }} {{- if $src }} {{- $img := printf `` ($src | htmlEscape) }} {{- $phrase := .Params.iconFor | default "" | htmlEscape }} {{- if and $phrase (strings.Contains $title $phrase) }} {{- $title = replace $title $phrase (print $img $phrase) 1 }} {{- else }} {{- $title = print $img $title }} {{- end }} {{- $title = print `` $title `` }} {{- end }} {{- $title | safeHTML -}}
